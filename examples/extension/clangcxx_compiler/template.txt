struct ShaderInterface {
template<typename T>
[[nodiscard]] static constexpr decltype(auto) materialize(T&& value) {
using value_type = ::std::remove_cvref_t<T>;
if constexpr (requires { typename value_type::EvalType; }) {
using eval_type = typename value_type::EvalType;
if constexpr (requires {
::std::forward<T>(value).operator eval_type();
}) {
return ::std::forward<T>(value).operator eval_type();
} else {
return ::std::forward<T>(value);
}
} else {
return ::std::forward<T>(value);
}
}

template<ShaderArgumentName Name, typename T>
[[nodiscard]] static constexpr auto arg(T&& value) {
using storage_type = ::std::conditional_t<
::std::is_lvalue_reference_v<T>, T, ::std::remove_cvref_t<T>>;
return ShaderNamedArgument<Name, storage_type>{::std::forward<T>(value)};
}

template<typename T>
static constexpr bool is_argument = requires {

};

template<ShaderArgumentName Name, typename T>
[[nodiscard]] static consteval bool argument_matches() noexcept {
if constexpr (is_argument<T>) {
return ::std::remove_cvref_t<T>::name == Name.value;
} else {
return false;
}
}

template<ShaderArgumentName Name, typename... Args>

(size_t{0u} + ... + size_t{argument_matches<Name, Args>()});

template<typename Lhs, typename Rhs>

if constexpr (is_argument<Lhs> && is_argument<Rhs>) {
return ::std::remove_cvref_t<Lhs>::name ==
::std::remove_cvref_t<Rhs>::name.value;
} else {
return false;
}
}

template<typename T, typename... Args>
static constexpr size_t equivalent_argument_count =
(size_t{0u} + ... + size_t{argument_names_match<T, Args>()});

template<typename... Args>
[[nodiscard]] static consteval bool arguments_unique() noexcept {
if constexpr ((is_argument<Args> && ...)) {
